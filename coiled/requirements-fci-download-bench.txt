# Environment for `benchmarks/fci_download_latency.py`.
#
# Deliberately tiny: the benchmark only exercises the EUMETSAT download path, which
# is `eumdac` and the standard library. Keeping torch and the rest of
# `requirements-stormscope-eu.txt` out means the per-region image build (Coiled builds
# one image per region) is a minute rather than the ~9.6 GB stormscope-eu build --
# which matters here precisely because the point is to run in a *new* region.
eumdac

# `coiled run` starts the VM as a one-node Dask cluster and runs the command inside
# that process, so the environment has to be able to host a scheduler. An env with
# eumdac alone dies at startup with
#   Cluster status is error (reason: Scheduler Stopped -> Software environment
#   exited with error code 1.)
# and no Python traceback in `coiled logs` -- the process never gets far enough to
# print one.
coiled
